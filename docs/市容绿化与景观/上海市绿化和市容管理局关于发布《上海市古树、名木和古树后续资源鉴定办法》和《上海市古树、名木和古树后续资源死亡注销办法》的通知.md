        function compareDate(invalidDate){

            if(invalidDate!=undefined&&invalidDate!=null&&invalidDate!=""){

                	//到期时间

                    let d2 = invalidDate;

                    //转换为标准时间"2019/12/29 23:59:59"

                    d2 = d2.replace(/-/g, "/");

                    d2 = new Date(d2);

                

                    //当前时间

                    let d1 = new Date();

                

                    //如果到期时间大于等于当前时间

                    //valueOf() 函数返回指定对象的原始值,这里返回以毫秒数存储的时间值

                    if (d2.valueOf() >= d1.valueOf()) {

                        $("#divInvalidDate").html("有效");

                    }

                    else{

                        $("#divInvalidDate").html("已失效");

                    }

            }

            else {

                $("#divInvalidDate").html("有效");

            }

        }
